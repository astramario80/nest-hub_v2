function displayTime(date) {
  return new Intl.DateTimeFormat([], { hour: 'numeric', minute: '2-digit' }).format(date);
}

function dayStamp() {
  const now = new Date();
  return [now.getFullYear(), String(now.getMonth() + 1).padStart(2, '0'), String(now.getDate()).padStart(2, '0')].join('-');
}

function escapeHtml(value) {
  return String(value)
    .replaceAll('&', '&amp;')
    .replaceAll('<', '&lt;')
    .replaceAll('>', '&gt;')
    .replaceAll('"', '&quot;')
    .replaceAll("'", '&#039;');
}

function parseCsv(text) {
  const rows = [];
  let row = [];
  let field = '';
  let quoted = false;
  for (let i = 0; i < text.length; i += 1) {
    const char = text[i];
    const next = text[i + 1];
    if (char === '"' && quoted && next === '"') {
      field += '"';
      i += 1;
    } else if (char === '"') {
      quoted = !quoted;
    } else if (char === ',' && !quoted) {
      row.push(field.trim());
      field = '';
    } else if ((char === '\n' || char === '\r') && !quoted) {
      if (char === '\r' && next === '\n') i += 1;
      row.push(field.trim());
      if (row.some(Boolean)) rows.push(row);
      row = [];
      field = '';
    } else {
      field += char;
    }
  }
  row.push(field.trim());
  if (row.some(Boolean)) rows.push(row);
  return rows;
}

function timeToday(value) {
  const parts = String(value).trim().split(':').map(Number);
  if (parts.length < 2 || parts.some(Number.isNaN)) return null;
  const date = new Date();
  date.setHours(parts[0], parts[1], parts[2] || 0, 0);
  return date;
}

function normalizeSchedule(csv) {
  const rows = parseCsv(csv).map(function (columns) {
    const label = columns[0] || '';
    const times = (columns[1] || '').split('-');
    const start = timeToday(times[0]);
    const end = timeToday(times[1]);
    const prep = columns[2] || '';
    const tag = columns[3] || '';
    const allText = [label, prep, tag].join(' ').toLowerCase();
    return {
      label: label,
      prep: prep,
      tag: tag,
      start: start,
      end: end,
      planning: allText.includes('planning'),
      lunch: allText.includes('lunch'),
      absent: label.trim().toLowerCase() === 'absent',
      silentLunch: false
    };
  }).filter(function (item) {
    const heading = item.label.toLowerCase();
    return item.label && item.start && item.end && heading !== 'period' && heading !== 'label';
  });

  rows.forEach(function (item, index) {
    if (!item.lunch) return;
    const previous = rows[index - 1];
    const next = rows[index + 1];
    item.silentLunch = Boolean((previous && previous.planning) || (next && next.planning));
  });
  return rows;
}

async function loadSchedule() {
  try {
    const response = await fetch(CONFIG.scheduleUrl, { cache: 'no-store' });
    if (!response.ok) throw new Error('Schedule returned ' + response.status);
    const text = await response.text();
    localStorage.setItem(STORAGE.schedule, text);
    state.schedule = normalizeSchedule(text);
  } catch (error) {
    const cached = localStorage.getItem(STORAGE.schedule);
    if (!cached) throw error;
    state.schedule = normalizeSchedule(cached);
    recordActivity('Using the most recently saved bell schedule.');
  }
  renderSchedule();
  return state.schedule;
}

function rowIcon(item) {
  if (item.silentLunch) return '🔇';
  if (item.planning) return '◌';
  if (item.lunch) return '◇';
  return '●';
}

function renderSchedule() {
  if (!state.schedule.length) {
    ui.scheduleList.innerHTML = '<div class="schedule-empty">No schedule is available for today.</div>';
    return;
  }
  ui.scheduleList.innerHTML = state.schedule.map(function (item) {
    const details = item.silentLunch ? 'Planning music · no lunch cues' : (item.prep || item.tag || 'Transition block');
    return '<article class="schedule-row" data-active="false">' +
      '<span class="row-icon">' + rowIcon(item) + '</span>' +
      '<div><strong>' + escapeHtml(item.label) + '</strong><span>' + escapeHtml(details) + '</span></div>' +
      '<time>' + escapeHtml(displayTime(item.start)) + '–' + escapeHtml(displayTime(item.end)) + '</time>' +
      '</article>';
  }).join('');
}

function completedForToday() {
  try {
    const saved = JSON.parse(localStorage.getItem(STORAGE.completed) || '{}');
    return saved.date === dayStamp() ? new Set(saved.ids || []) : new Set();
  } catch (error) {
    return new Set();
  }
}

function markCompleted(id) {
  const completed = completedForToday();
  completed.add(id);
  localStorage.setItem(STORAGE.completed, JSON.stringify({ date: dayStamp(), ids: Array.from(completed) }));
}

function isCompleted(id) {
  return completedForToday().has(id);
}

function transitionIsDue(item, next, now) {
  // Catch a delayed tick while this transition is still relevant, never over cleanup.
  const until = Math.max(next.start.getTime(), next.end.getTime() - CONFIG.cleanupLeadMs);
  return now >= item.end && now.getTime() < until;
}

function announcementPeriod(now) {
  return state.schedule.find(function (item) {
    return !item.silentLunch && isSecondPeriod(item) && now >= item.start && now < item.end &&
      now.getTime() < item.start.getTime() + CONFIG.announcementMs;
  });
}

function isSecondPeriod(item) {
  const label = item.label.toLowerCase();
  return label.includes('2nd') || label.includes('period 2');
}

function playlistFor(item) {
  const text = [item.label, item.prep, item.tag].join(' ').toLowerCase();
  if (item.planning) return CONFIG.quietUri;
  if (text.includes('fly day')) return CONFIG.flyDayUri;
  if (text.includes('planning') || text.includes('lunch') || text.includes('study support') || text.includes('dismissal')) {
    return CONFIG.quietUri;
  }
  if (text.includes('ctso')) return CONFIG.ctsoUri;
  for (let number = 0; number <= 7; number += 1) {
    if (new RegExp('(^|\\D)' + number + '(\\D|$)').test(item.label) ||
        item.label.toLowerCase().includes(number + ordinalSuffix(number))) {
      return CONFIG.periodUris[number];
    }
  }
  if (text.includes('advisory')) return CONFIG.periodUris[7];
  return '';
}

function ordinalSuffix(number) {
  if (number === 1) return 'st';
  if (number === 2) return 'nd';
  if (number === 3) return 'rd';
  return 'th';
}

function hasCleanup(item) {
  if (item.silentLunch) return false;
  if (!item.planning) return true;
  const next = state.schedule[state.schedule.indexOf(item) + 1];
  return Boolean(next && !next.planning && !next.lunch && !next.silentLunch);
}

function upcomingEvents(now) {
  const events = [];
  state.schedule.forEach(function (item, index) {
    const cleanup = new Date(item.end.getTime() - CONFIG.cleanupLeadMs);
    if (hasCleanup(item)) events.push({ at: cleanup, label: 'Cleanup · ' + item.label });
    const next = state.schedule[index + 1];
    if (next && !next.silentLunch && playlistFor(next)) {
      events.push({ at: next.planning ? next.start : item.end, label: 'Transition · ' + next.label });
    }
  });
  return events.filter(function (event) { return event.at >= now; })
    .sort(function (a, b) { return a.at - b.at; });
}

function updateReadouts(now) {
  const current = state.schedule.find(function (item) { return now >= item.start && now < item.end; });
  const rows = Array.from(ui.scheduleList.querySelectorAll('.schedule-row'));
  rows.forEach(function (row, index) {
    row.dataset.active = current === state.schedule[index] ? 'true' : 'false';
  });
  if (current) {
    ui.currentPeriod.textContent = current.label + (current.silentLunch ? ' · planning music' : '');
    ui.currentTime.textContent = displayTime(current.start) + '–' + displayTime(current.end);
  } else {
    ui.currentPeriod.textContent = 'No active block';
    ui.currentTime.textContent = 'Waiting for the next scheduled period';
  }
  const next = upcomingEvents(now)[0];
  ui.nextEvent.textContent = next ? next.label : 'Signals complete';
  ui.nextEventTime.textContent = next ? displayTime(next.at) : 'No more signals today';
  ui.lastCheck.textContent = displayTime(now);
}

function randomString(length) {
  const values = crypto.getRandomValues(new Uint8Array(length));
  return Array.from(values).map(function (value) { return (value % 36).toString(36); }).join('');
}

async function challengeFor(verifier) {
  const bytes = new TextEncoder().encode(verifier);
  const digest = await crypto.subtle.digest('SHA-256', bytes);
  return btoa(String.fromCharCode.apply(null, new Uint8Array(digest)))
    .replaceAll('+', '-').replaceAll('/', '_').replaceAll('=', '');
}

function tokenRecord() {
  try {
    return JSON.parse(localStorage.getItem(STORAGE.token) || 'null');
  } catch (error) {
    return null;
  }
}

async function refreshToken() {
  const saved = tokenRecord();
  const clientId = CONFIG.spotifyClientId;
  if (!saved || !saved.refresh_token || !clientId) return '';
  const body = new URLSearchParams({
    client_id: clientId,
    grant_type: 'refresh_token',
    refresh_token: saved.refresh_token
  });
  const response = await fetch('https://accounts.spotify.com/api/token', {
    method: 'POST',
    headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
    body: body
  });
  if (!response.ok) throw new Error('Spotify authorization expired');
  const updated = await response.json();
  const record = Object.assign({}, saved, updated, {
    expires_at: Date.now() + (updated.expires_in * 1000) - 60000
  });
  localStorage.setItem(STORAGE.token, JSON.stringify(record));
  return record.access_token;
}

async function accessToken(forceRefresh) {
  const saved = tokenRecord();
  if (!saved) return '';
  if (!forceRefresh && saved.access_token && saved.expires_at > Date.now()) return saved.access_token;
  return refreshToken();
}

async function beginSpotifyLogin() {
  const clientId = CONFIG.spotifyClientId;
  localStorage.setItem(STORAGE.clientId, clientId);
  const verifier = randomString(64);
  const oauthState = randomString(24);
  sessionStorage.setItem(STORAGE.verifier, verifier);
  sessionStorage.setItem(STORAGE.oauthState, oauthState);
  const params = new URLSearchParams({
    client_id: clientId,
    response_type: 'code',
    redirect_uri: redirectUri(),
    state: oauthState,
    scope: 'streaming user-read-private user-read-email user-read-playback-state user-read-currently-playing user-modify-playback-state',
    code_challenge_method: 'S256',
    code_challenge: await challengeFor(verifier)
  });
  window.top.location.href = 'https://accounts.spotify.com/authorize?' + params.toString();
}

async function handleSpotifyCallback() {
  const params = new URLSearchParams(window.location.search);
  const code = params.get('code');
  if (!code) return false;
  if (params.get('state') !== sessionStorage.getItem(STORAGE.oauthState)) {
    throw new Error('Spotify sign-in could not be verified. Please connect again.');
  }
  const verifier = sessionStorage.getItem(STORAGE.verifier);
  const clientId = CONFIG.spotifyClientId;
  const body = new URLSearchParams({
    client_id: clientId,
    grant_type: 'authorization_code',
    code: code,
    redirect_uri: redirectUri(),
    code_verifier: verifier
  });
  const response = await fetch('https://accounts.spotify.com/api/token', {
    method: 'POST',
    headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
    body: body
  });
  if (!response.ok) throw new Error('Spotify did not complete the connection.');
  const token = await response.json();
  token.expires_at = Date.now() + (token.expires_in * 1000) - 60000;
  localStorage.setItem(STORAGE.token, JSON.stringify(token));
  sessionStorage.removeItem(STORAGE.verifier);
  sessionStorage.removeItem(STORAGE.oauthState);
  history.replaceState({}, document.title, '/api/signals?asset=index');

  return true;
}

async function spotifyFetch(path, options, retry) {
  await checkNestAccess();
  const token = await accessToken(false);
  if (!token) throw new Error('Choose Spotify connection to connect an approved Premium account.');
  const request = Object.assign({}, options || {});
  request.headers = Object.assign({}, request.headers || {}, { Authorization: 'Bearer ' + token });
  const response = await fetch('https://api.spotify.com/v1' + path, request);
  if (response.status === 401 && retry !== false) {
    await accessToken(true);
    return spotifyFetch(path, options, false);
  }
  if (!response.ok && response.status !== 204) {
    let detail = '';
    try { detail = (await response.json()).error.message; } catch (error) { detail = response.statusText; }
    const error = new Error(detail || 'Spotify returned ' + response.status);
    error.status = response.status;
    error.spotifyPath = path;
    throw error;
  }
  return response;
}

function loadSpotifySdk() {
  return new Promise(function (resolve, reject) {
    if (window.Spotify) {
      resolve();
      return;
    }
    window.onSpotifyWebPlaybackSDKReady = resolve;
    const existing = document.querySelector('script[data-spotify-sdk]');
    if (existing) return;
    const script = document.createElement('script');
    script.src = 'https://sdk.scdn.co/spotify-player.js';
    script.dataset.spotifySdk = 'true';
    script.onerror = function () { reject(new Error('The Spotify player could not load.')); };
    document.head.appendChild(script);
  });
}

function preparePlayer() {
  if (state.player) return;
  if (!window.Spotify) throw new Error('The Spotify player is still loading. Try Start signals again.');
  state.player = new Spotify.Player({
    name: 'NEST™ Signals',
    getOAuthToken: async function (callback) {
      callback(await accessToken(false));
    },
    volume: 0.5
  });
  state.player.addListener('ready', function (details) {
    state.deviceId = details.device_id;
    state.spotifyReady = true;
    ui.audioStatus.textContent = 'Spotify · NEST player ready';
  });
  state.player.addListener('not_ready', function () {
    state.deviceId = '';
    state.spotifyReady = false;
    ui.audioStatus.textContent = 'Spotify player offline';
  });
  state.player.addListener('authentication_error', function () {
    playerFailed('Spotify authorization needs to be renewed. Choose Spotify connection to reconnect.');
  });
  state.player.addListener('account_error', function () {
    playerFailed('Spotify Premium is required for automatic web playback.');
  });
  state.player.addListener('autoplay_failed', function () {
    playerFailed('Your browser blocked audio. Return to this page and click Start signals to enable sound.');
  });
  state.player.addListener('initialization_error', function () {
    playerFailed('This browser could not create the Spotify audio player. Open Signals in a browser that supports Spotify playback.');
  });
  state.player.addListener('playback_error', function (details) {
    playerFailed('Spotify could not play the music: ' + details.message);
  });
}

function playerFailed(message) {
  state.playbackError = new Error(message);
  stopSignals();
  showError(state.playbackError);
}

async function connectPlayer() {
  preparePlayer();
  if (state.deviceId && state.spotifyReady) return;
  const connected = await state.player.connect();
  if (!connected) throw new Error('Spotify could not connect the classroom player.');
  const deadline = Date.now() + 12000;
  while (!state.deviceId && Date.now() < deadline) {
    await new Promise(function (resolve) { setTimeout(resolve, 200); });
  }
  if (state.playbackError) throw state.playbackError;
  if (!state.deviceId) throw new Error('Spotify did not make the classroom player available.');
}

function resetPlayerConnection() {
  if (state.player) state.player.disconnect();
  state.deviceId = '';
  state.spotifyReady = false;
}

async function waitForSpotifyDevice(active) {
  for (let attempt = 0; attempt < 12; attempt += 1) {
    if (state.playbackError) throw state.playbackError;
    if (!state.deviceId) return false;
    const response = await spotifyFetch('/me/player/devices');
    const data = await response.json();
    const device = (data.devices || []).find(function (entry) { return entry.id === state.deviceId; });
    if (device && device.is_restricted) throw new Error('Spotify is restricting control of the NEST player. Reconnect through Spotify connection.');
    if (device && (!active || device.is_active)) return true;
    await new Promise(function (resolve) { setTimeout(resolve, 500); });
  }
  return false;
}

async function activatePlayer() {
  try {
    // SDK readiness can precede registration with Spotify's playback API.
    if (!await waitForSpotifyDevice(false)) throw new Error('device-unavailable');
    for (let attempt = 0; attempt < 3; attempt += 1) {
      try {
        await spotifyFetch('/me/player', {
          method: 'PUT',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ device_ids: [state.deviceId], play: false })
        });
        if (await waitForSpotifyDevice(true)) return;
      } catch (error) {
        if (error.status !== 404) throw error;
      }
      await new Promise(function (resolve) { setTimeout(resolve, 750); });
    }
    throw new Error('device-unavailable');
  } catch (error) {
    resetPlayerConnection();
    if (error.message === 'device-unavailable' || error.status === 404) {
      throw new Error('Spotify could not find or activate this page’s NEST player. Keep this page open and click Start signals to reconnect. If it repeats, use Spotify connection to reconnect your account.');
    }
    throw error;
  }
}

async function setSpotifyVolume(level) {
  // Audio plays in this SDK player; avoid an additional device-targeted API call.
  if (state.player) await state.player.setVolume(level);
}

async function playbackState() {
  const response = await spotifyFetch('/me/player');
  if (response.status === 204) return null;
  return response.json();
}

async function playContext(uri, volume) {
  await activatePlayer();
  await setSpotifyVolume(volume);
  state.lastVolumeMode = '';
  await spotifyFetch('/me/player/play?device_id=' + encodeURIComponent(state.deviceId), {
    method: 'PUT',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ context_uri: uri })
  });
  // A successful API response does not mean browser audio actually started.
  for (let attempt = 0; attempt < 20; attempt += 1) {
    if (state.playbackError) throw state.playbackError;
    if (!state.running) throw new Error('Signals were stopped before playback was confirmed.');
    const playback = await state.player.getCurrentState();
    if (playback && !playback.paused && playback.context && playback.context.uri === uri) {
      try {
        await spotifyFetch('/me/player/shuffle?state=true&device_id=' + encodeURIComponent(state.deviceId), { method: 'PUT' });
      } catch (error) {
        recordActivity('Music is playing, but Spotify could not enable shuffle.');
      }
      return;
    }
    await new Promise(function (resolve) { setTimeout(resolve, 500); });
  }
  throw new Error('Spotify accepted the music request, but the NEST player did not start. Click Start signals to try again and allow audio in this browser.');
}

async function runCleanup(item) {
  const id = 'cleanup:' + item.end.getTime();
  if (isCompleted(id) || !hasCleanup(item)) return;
  const playback = await playbackState();
  if (playback && playback.is_playing && playback.context && playback.context.uri === CONFIG.cleanupUri && playback.device && playback.device.id === state.deviceId) {
    markCompleted(id);
    recordActivity('Cleanup was already playing, so it was left uninterrupted.');
    return;
  }
  await playContext(CONFIG.cleanupUri, 0.8);
  markCompleted(id);
  ui.audioStatus.textContent = 'Spotify · Cleanup playback confirmed';
  recordActivity('Cleanup music started for ' + item.label + '.');
}

async function runPlanning(item) {
  const id = String(item.start.getTime());
  if (state.planningBlock === id) return;
  const playback = await state.player.getCurrentState();
  if (!playback || playback.paused || !playback.context || playback.context.uri !== CONFIG.quietUri) {
    await playContext(CONFIG.quietUri, 0.5);
  }
  state.planningBlock = id;
  ui.audioStatus.textContent = 'Spotify · J&M Chill Folk playback confirmed';
  recordActivity('Planning period · J&M Chill Folk.');
}

async function runTransition(item, next) {
  const id = 'transition:' + item.end.getTime();
  if (isCompleted(id) || next.silentLunch) return;
  const uri = playlistFor(next);
  if (!uri) return;
  await playContext(uri, announcementPeriod(new Date()) ? 0 : 0.8);
  markCompleted(id);
  ui.audioStatus.textContent = 'Spotify · ' + next.label + ' playback confirmed';
  recordActivity('Music changed for ' + next.label + '.');
}

async function enforcePeriodVolume(now) {
  const current = state.schedule.find(function (item) { return now >= item.start && now < item.end; });
  if (!current) return;
  let mode = 'normal:' + current.start.getTime();
  let volume = 0.5;
  if (hasCleanup(current) && now >= new Date(current.end.getTime() - CONFIG.cleanupLeadMs)) {
    mode = 'cleanup:' + current.end.getTime();
    volume = 0.8;
  }
  if (isSecondPeriod(current) && now < new Date(current.start.getTime() + CONFIG.announcementMs)) {
    mode = 'announcements:' + current.start.getTime();
    volume = 0;
  }
  if (state.lastVolumeMode === mode) return;
  await setSpotifyVolume(volume);
  state.lastVolumeMode = mode;
  recordActivity(volume === 0 ? '2nd-period announcements are quiet.' : 'Period music volume set.');
}

let accessCheckedAt = typeof NEST_ACCESS_VERIFIED_UNTIL === 'number'
  ? Math.min(Date.now(), NEST_ACCESS_VERIFIED_UNTIL - 60000) : 0;
async function checkNestAccess(force) {
  if (!force && Date.now() - accessCheckedAt < 60000) return;
  let response;
  try {
    response = await fetch('/api/signals?asset=access', { credentials: 'same-origin', cache: 'no-store', signal: AbortSignal.timeout(55000) });
  } catch (_) { response = { ok: false, status: 503 }; }
  if (!response.ok) {
    stopSignals();
    if (state.player) { try { await state.player.pause(); } catch (_) {} state.player.disconnect(); }
    state.player = null; state.deviceId = ''; state.spotifyReady = false;
    if (response.status === 401 || response.status === 403) localStorage.removeItem(STORAGE.token);
    throw new Error(response.status === 503 ? 'NEST access could not be checked. Try Start signals again.' : 'NEST access has ended. Sign in to NEST with an authorized school account.');
  }
  accessCheckedAt = Date.now();
}

async function schedulerTick() {
  if (!state.running || state.busy) return;
  state.busy = true;
  try {
    await checkNestAccess();
    if (!state.running) return;
    const now = new Date();
    updateReadouts(now);
    if (state.schedule.some(function (item) { return item.absent; })) {
      setStatus('paused', 'ABSENT MODE', 'Signals are paused', 'The schedule contains an Absent row, so no music will be changed.');
      return;
    }
    const planning = state.schedule.find(function (item) { return (item.planning || item.silentLunch) && now >= item.start && now < item.end; });
    if (planning) {
      if (hasCleanup(planning) && now.getTime() >= planning.end.getTime() - CONFIG.cleanupLeadMs) {
        await runCleanup(planning);
      } else {
        await runPlanning(planning);
      }
    }
    for (let index = 0; !planning && index < state.schedule.length; index += 1) {
      const item = state.schedule[index];
      const cleanupTime = new Date(item.end.getTime() - CONFIG.cleanupLeadMs);
      if (now >= item.start && now >= cleanupTime && now < item.end) await runCleanup(item);
      const next = state.schedule[index + 1];
      if (next && !next.planning && transitionIsDue(item, next, now)) await runTransition(item, next);
    }
    await enforcePeriodVolume(now);
    if (!state.running || state.playbackError) return;
    const announcements = announcementPeriod(now);
    if (announcements) {
      const resumeAt = new Date(Math.min(announcements.end.getTime(), announcements.start.getTime() + CONFIG.announcementMs));
      setStatus('running', 'ANNOUNCEMENT QUIET TIME', 'Signals are active · music is muted', 'Second-period announcements: sound resumes at ' + displayTime(resumeAt) + '. Keep this page open.');
    } else {
      setStatus('running', 'SIGNALS ACTIVE', 'Classroom automation is running', 'Cleanup plays during the final six minutes of each scheduled period. Keep this page open and the computer awake.');
    }
  } catch (error) {
    stopSignals();
    showError(error);
  } finally {
    state.busy = false;
  }
}

async function requestWakeLock() {
  if (!('wakeLock' in navigator) || document.visibilityState !== 'visible') return;
  try {
    state.wakeLock = await navigator.wakeLock.request('screen');
  } catch (error) {
    recordActivity('Windows power settings must keep the laptop awake.');
  }
}

function startSignalsFromClick() {
  if (state.running || ui.start.disabled) return;
  if (!tokenRecord()) return startSignals();
  try {
    preparePlayer();
    state.playbackError = null;
    // Keep audio activation in the click itself, before awaiting any requests.
    const activation = state.player.activateElement();
    ui.start.disabled = true;
    return Promise.resolve(activation).then(startSignals).catch(function (error) {
      ui.start.disabled = false;
      showError(error);
    });
  } catch (error) {
    showError(error);
  }
}

async function startSignals() {
  try {
    ui.start.disabled = true;
    setStatus('working', 'CONNECTING', 'Preparing today’s signals', 'Loading the schedule and classroom Spotify player.');
    if (!tokenRecord()) {
      ui.dialog.showModal();
      throw new Error('Choose Connect Spotify to authorize an approved Premium account.');
    }
    await checkNestAccess();
    await loadSchedule();
    await loadProfile();
    if (state.profile && state.profile.product !== 'premium') {
      throw new Error('The connected Spotify account must be Premium.');
    }
    await connectPlayer();
    await activatePlayer();
    if (state.playbackError) throw state.playbackError;
    state.planningBlock = '';
    state.running = true;
    localStorage.setItem(STORAGE.running, 'true');
    ui.start.textContent = 'Signals running';
    ui.start.disabled = true;
    ui.stop.disabled = false;
    await requestWakeLock();
    await schedulerTick();
    if (!state.running) return;
    clearInterval(state.timer);
    state.timer = setInterval(schedulerTick, CONFIG.tickMs);
  } catch (error) {
    stopSignals();
    showError(error);
  }
}

function stopSignals() {
  state.running = false;
  localStorage.setItem(STORAGE.running, 'false');
  clearInterval(state.timer);
  state.timer = null;
  ui.start.textContent = 'Start signals';
  ui.start.disabled = false;
  ui.stop.disabled = true;
  if (state.wakeLock) state.wakeLock.release();
  setStatus('paused', 'STOPPED', 'Signals are stopped for today', 'Press Start signals whenever the room is ready again.');
}

function showError(error) {
  console.error(error);
  setStatus('error', 'NEEDS ATTENTION', 'Signals need a quick check', error.message || String(error));
}

function disconnectSpotify() {
  stopSignals();
  if (state.player) state.player.disconnect();
  state.player = null;
  state.deviceId = '';
  state.profile = null;
  localStorage.removeItem(STORAGE.token);
  ui.spotifyUser.textContent = 'Spotify is not connected';
  ui.spotifyAccountDetail.textContent = 'Premium is required for automatic playback.';
  ui.audioStatus.textContent = 'Spotify not connected';
}

function registerStatusTool() {
  if (!navigator.modelContext || typeof navigator.modelContext.registerTool !== 'function') return;
  navigator.modelContext.registerTool({
    name: 'read_signal_status',
    description: 'Read today’s classroom signal status without changing playback.',
    inputSchema: { type: 'object', properties: {}, additionalProperties: false },
    execute: async function () {
      const now = new Date();
      const current = state.schedule.find(function (item) { return now >= item.start && now < item.end; });
      const next = upcomingEvents(now)[0];
      return {
        content: [{
          type: 'text',
          text: JSON.stringify({
            running: state.running,
            currentBlock: current ? current.label : null,
            nextSignal: next ? next.label : null,
            nextSignalAt: next ? next.at.toISOString() : null,
            spotifyReady: state.spotifyReady
          })
        }]
      };
    }
  });
}

async function initialize() {
  ui.start.disabled = true;
  displayClock();
  setInterval(displayClock, 1000);
  ui.redirectUri.value = redirectUri();
  const previousClientId = localStorage.getItem(STORAGE.clientId);
  if (previousClientId && previousClientId !== CONFIG.spotifyClientId) {
    localStorage.removeItem(STORAGE.token);
    localStorage.setItem(STORAGE.running, 'false');
  }
  localStorage.setItem(STORAGE.clientId, CONFIG.spotifyClientId);

  ui.setup.addEventListener('click', function () { ui.dialog.showModal(); });
  ui.closeSetup.addEventListener('click', function () { ui.dialog.close(); });
  ui.form.addEventListener('submit', async function (event) {
    event.preventDefault();
    try { await beginSpotifyLogin(); } catch (error) { showError(error); }
  });
  ui.copyRedirect.addEventListener('click', async function () {
    await navigator.clipboard.writeText(redirectUri());
    ui.copyRedirect.textContent = 'Copied';
    setTimeout(function () { ui.copyRedirect.textContent = 'Copy'; }, 1500);
  });
  ui.disconnect.addEventListener('click', disconnectSpotify);
  ui.start.addEventListener('click', startSignalsFromClick);
  ui.stop.addEventListener('click', stopSignals);
  document.addEventListener('visibilitychange', function () {
    if (state.running && document.visibilityState === 'visible') requestWakeLock();
  });


  registerStatusTool();

  try {
    await checkNestAccess();
    await handleSpotifyCallback();
    if (tokenRecord()) {
      await loadSpotifySdk();
      preparePlayer();
      await loadProfile();
      ui.audioStatus.textContent = 'Spotify connected';
      setStatus('ready', 'READY', 'Spotify is connected', 'Confirm the Pyle speaker is selected, then start today’s signals.');
    }
    await loadSchedule();
    updateReadouts(new Date());
    if (localStorage.getItem(STORAGE.running) === 'true' && tokenRecord()) {
      setStatus('ready', 'READY', 'Click Start signals to enable audio', 'After reopening this page, a click is needed to enable sound in your browser.');
    }
  } catch (error) {
    showError(error);
  } finally {
    ui.start.disabled = false;
  }
}

initialize();
